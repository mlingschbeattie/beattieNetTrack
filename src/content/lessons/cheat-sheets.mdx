# Hide text in image LSB
def hide_text(image_path, text, output_path):
    img = Image.open(image_path)
    binary = ''.join(format(ord(c), '08b') for c in text) + '1111111111111110'
    pixels = list(img.getdata())
    
    for i in range(len(binary)):
        pixel = list(pixels[i])
        pixel[0] = pixel[0] & ~1 | int(binary[i])
        pixels[i] = tuple(pixel)
    
    img.putdata(pixels)
    img.save(output_path)

# Extract text from image LSB
def extract_text(image_path):
    img = Image.open(image_path)
    binary = ''
    
    for pixel in img.getdata():
        binary += str(pixel[0] & 1)
    
    chars = [chr(int(binary[i:i+8], 2)) for i in range(0, len(binary), 8)]
    return ''.join(chars).split('\\xff')[0]</div>
