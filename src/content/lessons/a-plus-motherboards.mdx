      import { aPlusMotherboardsQuestions } from "./data/a-plus-motherboards-questions.js";
      import { getProgress, updateProgress } from "./js/progressStore.js";

      const TRACK_ID = "pc-professional";
      const MODULE_ID = "a-plus-motherboards";

      const REVIEW_PRESETS = {
        "5min": 6,
        "15min": 15,
        "30min": 30
      };

      const DEFAULT_PRESET = "5min";

      let currentScore = 0;
      let questionsAnswered = 0;
      let totalQuestions = 0;
      let activePreset = DEFAULT_PRESET;

      function shuffleArray(array) {
        const shuffled = [...array];
        for (let i = shuffled.length - 1; i > 0; i--) {
          const j = Math.floor(Math.random() * (i + 1));
          [shuffled[i], shuffled[j]] = [shuffled[j], shuffled[i]];
        }
        return shuffled;
      }

      function buildReviewSet(presetKey) {
        const numQuestions = REVIEW_PRESETS[presetKey] || REVIEW_PRESETS[DEFAULT_PRESET];

        const easy = aPlusMotherboardsQuestions.filter(q => q.difficulty === "easy");
        const med = aPlusMotherboardsQuestions.filter(q => q.difficulty === "medium");
        const hard = aPlusMotherboardsQuestions.filter(q => q.difficulty === "hard");

        let pool;
        if (presetKey === "5min") {
          pool = [...easy, ...med];
        } else if (presetKey === "15min") {
          pool = [...easy, ...med, ...hard];
        } else {
          pool = [...med, ...hard];
        }

        if (pool.length === 0) {
          pool = [...aPlusMotherboardsQuestions];
        }

        const shuffled = shuffleArray(pool);
        return shuffled.slice(0, Math.min(numQuestions, shuffled.length));
      }

      function setActiveButton(presetKey) {
        document.querySelectorAll('.quiz-mode-controls button').forEach(btn => {
          btn.classList.toggle('active', btn.dataset.review === presetKey);
        });
      }

      function updateReviewModeNote(presetKey) {
        const note = document.getElementById('review-mode-note');
        const labelMap = {
          "5min": "5-minute",
          "15min": "15-minute",
          "30min": "30-minute"
        };
        const label = labelMap[presetKey] || labelMap[DEFAULT_PRESET];
        note.textContent = \`You are in \${label} review mode (\${totalQuestions} questions).\`;
      }

      function clearQuizContainer() {
        const container = document.getElementById('quiz-container');
        container.innerHTML = '';
        currentScore = 0;
        questionsAnswered = 0;
        totalQuestions = 0;
      }

      function loadProgress() {
        const progress = getProgress(TRACK_ID, MODULE_ID);

        if (progress.completed) {
          document.getElementById('progress-badge').style.display = 'block';
        }

        if (progress.attempts > 0) {
          document.getElementById('previous-score').style.display = 'block';
          document.getElementById('best-score-display').textContent = progress.bestScore;
          document.getElementById('attempts-display').textContent = progress.attempts;
        }

        return progress.lastPreset || DEFAULT_PRESET;
      }

      function saveProgress() {
        const progress = getProgress(TRACK_ID, MODULE_ID);
        const newBestScore = Math.max(progress.bestScore, currentScore);
        const isCompleted = totalQuestions > 0 && currentScore === totalQuestions;

        updateProgress(TRACK_ID, MODULE_ID, {
          attempts: progress.attempts + 1,
          bestScore: newBestScore,
          completed: isCompleted || progress.completed,
          lastVisited: Date.now(),
          lastPreset: activePreset
        });

        if (isCompleted) {
          document.getElementById('progress-badge').style.display = 'block';
        }
        document.getElementById('previous-score').style.display = 'block';
        document.getElementById('best-score-display').textContent = newBestScore;
        document.getElementById('attempts-display').textContent = progress.attempts + 1;
      }

      function generateMoboQuiz(presetKey = DEFAULT_PRESET) {
        const selectedQuestions = buildReviewSet(presetKey);
        const container = document.getElementById('quiz-container');
        totalQuestions = selectedQuestions.length;

        if (totalQuestions === 0) {
          container.innerHTML = '<div class="info-box">No questions available for this review mode.</div>';
          return;
        }

        selectedQuestions.forEach((q, index) => {
          const shuffledOptions = q.options.map((opt, idx) => ({ text: opt, isCorrect: idx === q.correctIndex }));
          const randomized = shuffleArray(shuffledOptions);

          const questionDiv = document.createElement('div');
          questionDiv.className = 'quiz-question';
          questionDiv.setAttribute('data-question', index + 1);

          const questionText = document.createElement('p');
          questionText.className = 'question-text';
          questionText.textContent = q.stem;
          questionDiv.appendChild(questionText);

          const optionsDiv = document.createElement('div');
          optionsDiv.className = 'quiz-options';

          randomized.forEach(option => {
            const btn = document.createElement('button');
            btn.className = 'quiz-option';
            btn.textContent = option.text;
            btn.dataset.correct = option.isCorrect;
            optionsDiv.appendChild(btn);
          });

          questionDiv.appendChild(optionsDiv);

          const feedback = document.createElement('div');
          feedback.className = 'quiz-feedback';
          questionDiv.appendChild(feedback);

          container.appendChild(questionDiv);
        });

        const scoreDiv = document.createElement('div');
        scoreDiv.className = 'quiz-score';
        scoreDiv.innerHTML = 'Score: <span id="score">0</span>/' + totalQuestions;
        container.appendChild(scoreDiv);

        document.querySelectorAll('.quiz-option').forEach(btn => {
          btn.addEventListener('click', handleMoboAnswer);
        });
      }

      function handleMoboAnswer() {
        const question = this.closest('.quiz-question');
        const options = question.querySelectorAll('.quiz-option');
        const feedback = question.querySelector('.quiz-feedback');
        const isCorrect = this.dataset.correct === 'true';

        options.forEach(opt => {
          opt.disabled = true;
          if (opt.dataset.correct === 'true') opt.classList.add('correct');
        });

        if (isCorrect) {
          this.classList.add('correct');
          feedback.textContent = '✓ Correct!';
          feedback.style.color = '#00ff88';
          currentScore++;
        } else {
          this.classList.add('incorrect');
          feedback.textContent = '✗ Incorrect.';
          feedback.style.color = '#ff4444';
        }

        questionsAnswered++;
        updateMoboScore();

        if (questionsAnswered === totalQuestions) {
          saveProgress();
        }
      }

      function updateMoboScore() {
        document.getElementById('score').textContent = currentScore;
      }

      function setReviewMode(presetKey, persist = true) {
        activePreset = REVIEW_PRESETS[presetKey] ? presetKey : DEFAULT_PRESET;
        clearQuizContainer();
        generateMoboQuiz(activePreset);
        setActiveButton(activePreset);
        updateReviewModeNote(activePreset);

        if (persist) {
          updateProgress(TRACK_ID, MODULE_ID, {
            lastPreset: activePreset,
            lastVisited: Date.now()
          });
        }
      }

      function bindReviewButtons() {
        document.querySelectorAll('.quiz-mode-controls button').forEach(btn => {
          btn.addEventListener('click', () => {
            setReviewMode(btn.dataset.review);
          });
        });
      }

      document.addEventListener('DOMContentLoaded', () => {
        const initialPreset = loadProgress();
        bindReviewButtons();
        setReviewMode(initialPreset, false);
      });
