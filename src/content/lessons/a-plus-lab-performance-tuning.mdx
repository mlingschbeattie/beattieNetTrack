      function calculateMetrics() {
        let bootTime = activeScenario.initialMetrics.bootTime;
        let cpuUsage = activeScenario.initialMetrics.cpuUsage;
        let ramUsage = activeScenario.initialMetrics.ramUsage;
        let diskUsage = activeScenario.initialMetrics.diskUsage;
        
        // Calculate reductions from disabled items
        Object.values(startupStates).forEach(prog => {
          if (!prog.enabled) {
            bootTime -= prog.impact * 0.6;
            cpuUsage -= prog.impact * 0.3;
            ramUsage -= prog.impact * 0.4;
            diskUsage -= prog.impact * 0.5;
          }
        });
        
        Object.values(serviceStates).forEach(svc => {
          if (!svc.enabled) {
            cpuUsage -= svc.impact * 0.4;
            ramUsage -= svc.impact * 0.5;
            diskUsage -= svc.impact * 0.3;
          }
        });
        
        Object.values(effectStates).forEach(eff => {
          if (!eff.enabled) {
            cpuUsage -= eff.impact * 0.3;
            ramUsage -= eff.impact * 0.2;
          }
        });
        
        return {
          bootTime: Math.max(15, Math.round(bootTime)),
          cpuUsage: Math.max(5, Math.round(cpuUsage)),
          ramUsage: Math.max(20, Math.round(ramUsage)),
          diskUsage: Math.max(5, Math.round(diskUsage))
        };
      }

      function evaluatePerformance() {
        const newMetrics = calculateMetrics();
        const target = activeScenario.targetMetrics;
        const feedback = [];
        let score = 0;
        
        // Check if essential items were disabled (penalties)
        let essentialDisabled = 0;
        Object.values(startupStates).forEach(prog => {
          if (prog.essential && !prog.enabled) {
            essentialDisabled++;
            feedback.push({
              type: 'fail',
              text: \`✗ You disabled "\${prog.name}" which is essential for system functionality (-25 points)\`
            });
          }
        });
        
        Object.values(serviceStates).forEach(svc => {
          if (svc.essential && !svc.enabled) {
            essentialDisabled++;
            feedback.push({
              type: 'fail',
              text: \`✗ You disabled "\${svc.name}" which is essential for system functionality (-25 points)\`
            });
          }
        });
        
        score -= essentialDisabled * 25;
        
        // Check performance improvements
        const bootImprovement = ((activeScenario.initialMetrics.bootTime - newMetrics.bootTime) / activeScenario.initialMetrics.bootTime) * 100;
        const cpuImprovement = ((activeScenario.initialMetrics.cpuUsage - newMetrics.cpuUsage) / activeScenario.initialMetrics.cpuUsage) * 100;
        const ramImprovement = ((activeScenario.initialMetrics.ramUsage - newMetrics.ramUsage) / activeScenario.initialMetrics.ramUsage) * 100;
        const diskImprovement = ((activeScenario.initialMetrics.diskUsage - newMetrics.diskUsage) / activeScenario.initialMetrics.diskUsage) * 100;
        
        if (newMetrics.bootTime <= target.bootTime + 10) {
          score += 25;
          feedback.push({
            type: 'pass',
            text: \`✓ Boot time improved from \${activeScenario.initialMetrics.bootTime}s to \${newMetrics.bootTime}s (\${Math.round(bootImprovement)}% improvement) (+25 points)\`
          });
        } else {
          feedback.push({
            type: 'fail',
            text: \`✗ Boot time only improved to \${newMetrics.bootTime}s, target was \${target.bootTime}s (0 points)\`
          });
        }
        
        if (newMetrics.cpuUsage <= target.cpuUsage + 5) {
          score += 25;
          feedback.push({
            type: 'pass',
            text: \`✓ CPU usage reduced from \${activeScenario.initialMetrics.cpuUsage}% to \${newMetrics.cpuUsage}% (\${Math.round(cpuImprovement)}% improvement) (+25 points)\`
          });
        } else {
          feedback.push({
            type: 'fail',
            text: \`✗ CPU usage only reduced to \${newMetrics.cpuUsage}%, target was \${target.cpuUsage}% (0 points)\`
          });

        if (newMetrics.ramUsage <= target.ramUsage + 10) {
          score += 25;
          feedback.push({
            type: 'pass',
            text: \`✓ RAM usage reduced from \${activeScenario.initialMetrics.ramUsage}% to \${newMetrics.ramUsage}% (\${Math.round(ramImprovement)}% improvement) (+25 points)\`

            type: 'fail',
            text: \`✗ RAM usage only reduced to \${newMetrics.ramUsage}%, target was \${target.ramUsage}% (0 points)\`
