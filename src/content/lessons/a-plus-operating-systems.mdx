      function shuffleArray(array) {
        const shuffled = [...array];
        for (let i = shuffled.length - 1; i > 0; i--) {
          const j = Math.floor(Math.random() * (i + 1));
          [shuffled[i], shuffled[j]] = [shuffled[j], shuffled[i]];
        }
        return shuffled;
      }

      function buildReviewSet(presetKey) {
        const numQuestions = REVIEW_PRESETS[presetKey] || REVIEW_PRESETS[DEFAULT_PRESET];

        const easy = aPlusOperatingSystemsQuestions.filter(q => q.difficulty === "easy");
        const med = aPlusOperatingSystemsQuestions.filter(q => q.difficulty === "medium");
        const hard = aPlusOperatingSystemsQuestions.filter(q => q.difficulty === "hard");

        let pool;
        if (presetKey === "5min") {
          pool = [...easy, ...med];
        } else if (presetKey === "15min") {
          pool = [...easy, ...med, ...hard];
        } else {
          pool = [...med, ...hard];
        }

        if (pool.length === 0) {
          pool = [...aPlusOperatingSystemsQuestions];
        }

        const shuffled = shuffleArray(pool);
        return shuffled.slice(0, Math.min(numQuestions, shuffled.length));
      }
