      function updateBudget() {
        const budget = activeScenario.budget;
        const spent = selectedUpgrades.reduce((total, upgradeId) => {
          const upgrade = activeScenario.upgradeOptions.find(u => u.id === upgradeId);
          return total + (upgrade?.price || 0);
        }, 0);
        const remaining = budget - spent;
        
        document.getElementById('budget-amount').textContent = \`$\${budget}\`;
        document.getElementById('budget-spent').innerHTML = \`Spent: <span style="color: \${spent > budget ? '#ff4444' : '#ffaa00'}">$\${spent}</span> | Remaining: <span style="color: \${remaining < 0 ? '#ff4444' : '#00ff88'}">$\${remaining}</span>\`;
      }

      function evaluateUpgradePlan() {
        const feedback = [];
        let score = 0;
        
        const budget = activeScenario.budget;
        const spent = selectedUpgrades.reduce((total, upgradeId) => {
          const upgrade = activeScenario.upgradeOptions.find(u => u.id === upgradeId);
          return total + (upgrade?.price || 0);
        }, 0);
        
        // Check budget
        if (spent > budget) {
          feedback.push({
            type: 'fail',
            text: \`✗ You exceeded the budget by $\${spent - budget} (-50 points)\`
          });
          score -= 50;
        } else {
          feedback.push({
            type: 'pass',
            text: \`✓ Stayed within budget ($\${spent}/$\${budget}) (+20 points)\`
          });
          score += 20;
        }
        
        // Check if optimal upgrades were selected
        const optimalUpgrades = activeScenario.optimalUpgrades;
        const selectedOptimal = optimalUpgrades.filter(id => selectedUpgrades.includes(id));
        const missedOptimal = optimalUpgrades.filter(id => !selectedUpgrades.includes(id));
        
        selectedOptimal.forEach(upgradeId => {
          const upgrade = activeScenario.upgradeOptions.find(u => u.id === upgradeId);
          feedback.push({
            type: 'pass',
            text: \`✓ Correctly identified \${upgrade.name} as a priority upgrade (+\${40 / optimalUpgrades.length} points)\`
          });
          score += 40 / optimalUpgrades.length;
        });
        
        missedOptimal.forEach(upgradeId => {
          const upgrade = activeScenario.upgradeOptions.find(u => u.id === upgradeId);
          feedback.push({
            type: 'fail',
            text: \`✗ Missed priority upgrade: \${upgrade.name} (0 points)\`
          });
        });
        
        // Check for unnecessary upgrades
        const unnecessary = selectedUpgrades.filter(id => !optimalUpgrades.includes(id));
        if (unnecessary.length > 0) {
          unnecessary.forEach(upgradeId => {
            const upgrade = activeScenario.upgradeOptions.find(u => u.id === upgradeId);
            feedback.push({
              type: 'warn',
              text: \`⚠️ \${upgrade.name} was selected but lower priority for this use case\`
            });
          });
        }
        
        // Bonus for efficiency
        if (spent < budget && selectedOptimal.length === optimalUpgrades.length) {
          const efficiency = ((budget - spent) / budget) * 100;
          const bonusPoints = Math.round(efficiency / 5);
          feedback.push({
            type: 'pass',
            text: \`✓ Efficient spending! Saved $\${budget - spent} (+\${bonusPoints} bonus points)\`
          });
          score += bonusPoints;
        }
        
        // Total impact analysis
        const totalImpact = selectedUpgrades.reduce((total, upgradeId) => {
          const upgrade = activeScenario.upgradeOptions.find(u => u.id === upgradeId);
          return total + (upgrade?.impact || 0);
        }, 0);
        
        feedback.push({
          type: 'pass',
          text: \`📊 Total performance improvement: +\${totalImpact}%\`
        });
        
        return { score: Math.max(0, Math.min(100, Math.round(score))), feedback };
      }
