    <script type="module">
      import { generalSkillsQuestions } from "./data/general-skills-questions.js";
      import { getProgress, updateProgress } from "./js/progressStore.js";
      import { QuizEngine } from "./js/quizEngine.js";
      import { convertToQuizFormat, buildReviewSet } from "./js/quizAdapter.js";

      const TRACK_ID = "cybersecurity";
      const MODULE_ID = "general-skills";

      const REVIEW_PRESETS = {
        "5min": 6,
        "15min": 15,
        "30min": 30
      };

      const DEFAULT_PRESET = "5min";

      let currentQuiz = null;
      let activePreset = DEFAULT_PRESET;

      const cmdChallenges = [
        { task: 'Find all .txt files in the current directory', answer: 'find . -name "*.txt"', hint: 'Use find with -name flag' },
        { task: 'Search for "error" in log.txt', answer: 'grep "error" log.txt', hint: 'Use grep command' },
        { task: 'Make script.sh executable', answer: 'chmod +x script.sh', hint: 'Use chmod to change permissions' },
        { task: 'Show the last 20 lines of syslog', answer: 'tail -n 20 syslog', hint: 'Use tail with -n flag' },
        { task: 'Count lines in data.csv', answer: 'wc -l data.csv', hint: 'Use wc with -l flag' },
        { task: 'List files sorted by size', answer: 'ls -lS', hint: 'Use ls with -l and -S flags' }
      ];
      let currentCmd = 0;

      function loadProgress() {
        const progress = getProgress(TRACK_ID, MODULE_ID);

        if (progress.completed) {
          document.getElementById('progress-badge').style.display = 'block';
        }

        if (progress.attempts > 0) {
          document.getElementById('previous-score').style.display = 'block';
          document.getElementById('best-score-display').textContent = progress.bestScore;
          document.getElementById('attempts-display').textContent = progress.attempts;
        }

        return progress.lastPreset || DEFAULT_PRESET;
      }

      function saveProgress(score, total) {
        const progress = getProgress(TRACK_ID, MODULE_ID);
        const newBestScore = Math.max(progress.bestScore, score);
        const isCompleted = total > 0 && score === total;

        updateProgress(TRACK_ID, MODULE_ID, {
          attempts: progress.attempts + 1,
          bestScore: newBestScore,
          completed: isCompleted || progress.completed,
          lastVisited: Date.now(),
          lastPreset: activePreset
        });

        if (isCompleted) {
          document.getElementById('progress-badge').style.display = 'block';
          if (window.showBadge) window.showBadge('Module Mastered!', '🏆');
        }
        document.getElementById('previous-score').style.display = 'block';
        document.getElementById('best-score-display').textContent = newBestScore;
        document.getElementById('attempts-display').textContent = progress.attempts + 1;
      }

      function generateQuiz(presetKey = DEFAULT_PRESET) {
        const selectedQuestions = buildReviewSet(generalSkillsQuestions, presetKey, REVIEW_PRESETS);
        const quizQuestions = convertToQuizFormat(selectedQuestions);

        currentQuiz = new QuizEngine('quiz-container', quizQuestions, {
          onCorrect: () => { if (window.addXp) window.addXp(5); },
          onIncorrect: () => {},
          showLetters: true
        });

        const originalFinish = currentQuiz.finishQuiz.bind(currentQuiz);
        currentQuiz.finishQuiz = function() {
          originalFinish();
          const state = this.getState();
          saveProgress(state.score, state.totalQuestions);
          if (window.burstConfetti) window.burstConfetti();
        };

        currentQuiz.render();

        const labelMap = { "5min": "5-minute", "15min": "15-minute", "30min": "30-minute" };
        const label = labelMap[presetKey] || labelMap[DEFAULT_PRESET];
        document.getElementById('review-mode-note').textContent = \`You are in \{label} review mode (\{quizQuestions.length} questions).\`;

        document.querySelectorAll('.quiz-mode-controls button').forEach(btn => {
          btn.classList.toggle('active', btn.dataset.review === presetKey);
        });
      }

      function bindReviewButtons() {
        document.querySelectorAll('.quiz-mode-controls button').forEach(btn => {
          btn.addEventListener('click', () => {
            activePreset = btn.dataset.review;
            generateQuiz(activePreset);
            updateProgress(TRACK_ID, MODULE_ID, {
              lastPreset: activePreset,
              lastVisited: Date.now()
            });
          });
        });
      }

      function loadChallenge() {
        document.getElementById('cmd-scenario').textContent = cmdChallenges[currentCmd].task;
        document.getElementById('cmd-builder-input').value = '';
        document.getElementById('cmd-result').textContent = '';
      }

      function bindLabControls() {
        document.getElementById('check-cmd').addEventListener('click', () => {
          const userCmd = document.getElementById('cmd-builder-input').value.trim();
          const correct = cmdChallenges[currentCmd].answer;
          const result = document.getElementById('cmd-result');

          if (userCmd === correct) {
            result.innerHTML = \`<strong style="color: #00ff88;">✓ Correct!</strong><br><br>Command: <code>\${correct}</code>\`;
          } else {
            result.innerHTML = \`<strong style="color: #ff4444;">✗ Not quite.</strong><br><br>Expected: <code>\{correct}</code><br>You entered: <code>\{userCmd}</code>\`;
          }
        });

        document.getElementById('show-hint').addEventListener('click', () => {
          document.getElementById('cmd-result').innerHTML = \`<strong>Hint:</strong> \${cmdChallenges[currentCmd].hint}\`;
        });

        document.getElementById('next-cmd').addEventListener('click', () => {
          currentCmd = (currentCmd + 1) % cmdChallenges.length;
          loadChallenge();
        });
      }

      document.addEventListener('DOMContentLoaded', () => {
        const initialPreset = loadProgress();
        bindReviewButtons();
        generateQuiz(initialPreset);
        loadChallenge();
        bindLabControls();
      });
    </script>
